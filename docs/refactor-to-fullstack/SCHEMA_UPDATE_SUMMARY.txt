================================================================================================
  ⚠️ IMPORTANT: personal_id Caching Strategy
================================================================================================

DO NOT query the database every time! Use caching:

// CLIENT-SIDE: When fetching list data
const response = await fetch('/api/transactions');
const { data, meta } = await response.json();

// Cache the max personal_id
localStorage.setItem('max_transaction_personal_id', meta.max_personal_id.toString());

// When creating a new record
const cachedMax = parseInt(localStorage.getItem('max_transaction_personal_id') || '0');
const nextPersonalId = cachedMax + 1;

// Send personal_id from client
await fetch('/api/transactions', {
  method: 'POST',
  body: JSON.stringify({
    personal_id: nextPersonalId,
    // ... other fields
  }),
});

// Update cache
localStorage.setItem('max_transaction_personal_id', nextPersonalId.toString());

This is CRITICAL for performance and avoiding unnecessary database queries!

// CLIENT-SIDE: In your component
import { transactionService } from '@/services/transactionService';

const createTransaction = async () => {
  // Get next personal_id from cache
  const cachedMax = parseInt(localStorage.getItem('max_transaction_personal_id') || '0');
  const nextPersonalId = cachedMax + 1;
  
  const transaction = await transactionService.create({
    personal_id: nextPersonalId, // Send from client
    date: '2025-01-15',
    account_id: 'account-uuid',
    category_id: 'category-uuid',
    amount: 50000,
    type: 'EXPENSE',
    note: 'Grocery shopping',
    position: null, // Always null for now
  });
  
  // Update cache
  localStorage.setItem('max_transaction_personal_id', nextPersonalId.toString());
};

// SERVER-SIDE: In your Route Handler (app/api/transactions/route.ts)
export async function POST(request: NextRequest) {
  const userId = await getCurrentUserId(request);
  const body = await request.json();
  
  const transaction = await db.transactions.create({
    data: {
      id: crypto.randomUUID(),
      user_id: userId,
      personal_id: body.personal_id, // From client cache
      date: body.date,
      account_id: body.account_id,
      category_id: body.category_id,
      amount: body.amount,
      type: body.type,
      note: body.note,
      position: null, // Always null
      created_at: new Date(),
      created_by: userId,
    },
  });
  
  return NextResponse.json(transaction, { status: 201 });
}

// GET endpoint returns max for caching
export async function GET(request: NextRequest) {
  const userId = await getCurrentUserId(request);
  const transactions = await db.transactions.findMany({
    where: { user_id: userId },
    orderBy: { personal_id: 'desc' },
  });
  
  return NextResponse.json({
    data: transactions,
    meta: { max_personal_id: transactions[0]?.personal_id || 0 },
  });
}
